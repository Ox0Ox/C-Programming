### Parameter Passing

#### Pass by Value

In C, all parameters are passed by value, meaning a copy of the argument is passed to the function.

```c
#include <stdio.h>

void modifyValue(int x) {
    x = x * 2;  // Modifies the local copy, not the original
    printf("Inside function: x = %d\n", x);
}

int main() {
    int num = 10;
    
    printf("Before function call: num = %d\n", num);
    modifyValue(num);
    printf("After function call: num = %d\n", num);  // num remains unchanged
    
    return 0;
}
```

#### Pass by Reference (using pointers)

To modify the original value, pass a pointer to the variable.

```c
#include <stdio.h>

void modifyValue(int *x) {
    *x = *x * 2;  // Modifies the original value
    printf("Inside function: *x = %d\n", *x);
}

int main() {
    int num = 10;
    
    printf("Before function call: num = %d\n", num);
    modifyValue(&num);  // Pass the address of num
    printf("After function call: num = %d\n", num);  // num is modified
    
    return 0;
}
```

void toUpperCase(char *text) {

void toLowerCase(char *text) {

// Function to capitalize first letter of each word
void capitalizeWords(char *text) {
    int capitalize = 1;  // Flag to indicate if next character should be capitalized
    
    while (*text) {
        if (isspace(*text)) {
            capitalize = 1;
        } else if (capitalize) {
            *text = toupper(*text);
            capitalize = 0;
        } else {
            *text = tolower(*text);
        }
        text++;
    }

// Function to reverse the text
void reverseText(char *text) {
    char *start = text;
    char *end = text + strlen(text) - 1;
    
    while (start < end) {
        // Swap characters
        char temp = *start;
        *start = *end;
        *end = temp;
        
        // Move pointers
        start++;
        end--;
    }
}

## Storage Classes

Storage classes in C determine the scope, lifetime, and initialization of variables.

### auto Storage Class

The `auto` storage class is the default for local variables. Variables are automatically created at the beginning of a block and destroyed at the end.

```c
#include <stdio.h>

void demoAutoStorage() {
    auto int x = 10;  // 'auto' is optional, as it's the default
    printf("Inside function: x = %d\n", x);
}

int main() {
    int y = 20;  // 'auto' is implied
    {
        auto int z = 30;
        printf("Inside block: y = %d, z = %d\n", y, z);
    }
    // z is no longer accessible here
    
    demoAutoStorage();
    // x from demoAutoStorage() is not accessible here
    
    return 0;
}
```

### register Storage Class

The `register` storage class suggests the compiler to store the variable in a CPU register for faster access. However, modern compilers optimize well, so this is often unnecessary.

```c
#include <stdio.h>
#include <time.h>

int main() {
    // Example with register variable for loop counter
    clock_t start, end;
    double cpu_time_used;
    
    start = clock();
    register int i;  // Suggest compiler to use a CPU register
    int sum = 0;
    for (i = 0; i < 1000000; i++) {
        sum += i;
    }
    end = clock();
    cpu_time_used = ((double) (end - start)) / CLOCKS_PER_SEC;
    
    printf("Sum: %d\n", sum);
    printf("Time taken: %f seconds\n", cpu_time_used);
    
    return 0;
}
```

### static Storage Class

The `static` storage class has two uses:
1. For local variables, it preserves the value between function calls.
2. For global variables and functions, it limits the scope to the file.

```c
#include <stdio.h>

// Function using static variable
void countCalls() {
    static int count = 0;  // Initialized only once
    count++;
    printf("This function has been called %d time(s)\n", count);
}

// Static global variable
static int fileScope = 100;  // Only accessible within this file

int main() {
    printf("Initial value of fileScope: %d\n", fileScope);
    
    // Demonstrate static local variable
    for (int i = 0; i < 5; i++) {
        countCalls();
    }
    
    return 0;
}
```

The `extern` storage class declares a variable that is defined elsewhere (typically in another file).

## Callback

A callback is a function that is passed as an argument to another function and is invoked at a specific time or when a specific event occurs.

### Basic Callback Example

```c
#include <stdio.h>

// Callback function type
typedef void (*CallbackFunc)(int);

// Functions that will be used as callbacks
void printSquare(int num) {
    printf("Square of %d is %d\n", num, num * num);
}

void printCube(int num) {
    printf("Cube of %d is %d\n", num, num * num * num);
}

// Function that takes a callback
void processNumbers(int arr[], int size, CallbackFunc callback) {
    for (int i = 0; i < size; i++) {
        callback(arr[i]);
    }
}

int main() {
    int numbers[] = {1, 2, 3, 4, 5};
    int size = sizeof(numbers) / sizeof(numbers[0]);
    
    printf("Processing with square callback:\n");
    processNumbers(numbers, size, printSquare);
    
    printf("\nProcessing with cube callback:\n");
    processNumbers(numbers, size, printCube);
    
    return 0;
}
```

### Advanced Callback Example with Context

#include <stdio.h>

void powerOf(void *exponent, int value) {
    int exp = *((int *)exponent);
    int result = 1;
    for (int i = 0; i < exp; i++) {
        result *= value;
    }
    printf("%d^%d = %d\n", value, exp, result);
}

// Function that uses the callback info
void processWithContext(int arr[], int size, CallbackInfo cbInfo) {
    for (int i = 0; i < size; i++) {
        cbInfo.callback(cbInfo.data, arr[i]);
    }
}

int main() {
    int numbers[] = {1, 2, 3, 4, 5};
    int size = sizeof(numbers) / sizeof(numbers[0]);
    
    // Set up callback with context for multiplication
    int factor = 10;
    CallbackInfo multiplyInfo = {multiplyBy, &factor};
    
    printf("Multiplying by %d:\n", factor);
    processWithContext(numbers, size, multiplyInfo);
    
    // Set up callback with context for exponentiation
    int exponent = 3;
    CallbackInfo powerInfo = {powerOf, &exponent};
    
    printf("\nRaising to power %d:\n", exponent);
    processWithContext(numbers, size, powerInfo);
    
    return 0;
}

## Recursion

Recursion is a technique where a function calls itself to solve a problem.

### Basic Recursion Example: Factorial

```c
#include <stdio.h>

// Recursive function to calculate factorial
unsigned long factorial(int n) {
    // Base case
    if (n == 0 || n == 1) {
        return 1;
    }
    
    // Recursive case
    return n * factorial(n - 1);
}

int main() {
    for (int i = 0; i <= 10; i++) {
        printf("Factorial of %d = %lu\n", i, factorial(i));
    }
    
    return 0;
}
```
