### Advanced Example: Map-Reduce with Callbacks

Let's implement a simple Map-Reduce pattern using callbacks:

```c
#include <stdio.h>
#include <stdlib.h>

// Map function: applies a transformation to each element in an array
int* mapArray(int arr[], int size, int (*transform)(int)) {
    int* result = (int*)malloc(size * sizeof(int));
    for (int i = 0; i < size; i++) {
        result[i] = transform(arr[i]);
    }
    return result;
}

// Reduce function: reduces an array to a single value using an accumulator
int reduceArray(int arr[], int size, int (*combine)(int, int), int initialValue) {
    int accumulator = initialValue;
    for (int i = 0; i < size; i++) {
        accumulator = combine(accumulator, arr[i]);
    }
    return accumulator;
}

// Transform functions for map
int square(int x) {
    return x * x;
}

int double_value(int x) {
    return x * 2;
}

int absolute(int x) {
    return x < 0 ? -x : x;
}

// Combine functions for reduce
int sum(int acc, int value) {
    return acc + value;
}

int product(int acc, int value) {
    return acc * value;
}

int max(int acc, int value) {
    return acc > value ? acc : value;
}

// Function to print an array
void printArray(int arr[], int size) {
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

int main() {
    int arr[] = {-3, 2, 0, 7, -5, 11, 4};
    int size = sizeof(arr) / sizeof(arr[0]);
    
    printf("Original array: ");
    printArray(arr, size);
    
    // Map examples
    int* squares = mapArray(arr, size, square);
    printf("Squares: ");
    printArray(squares, size);
    
    int* doubles = mapArray(arr, size, double_value);
    printf("Doubled values: ");
    printArray(doubles, size);
    
    int* absolutes = mapArray(arr, size, absolute);
    printf("Absolute values: ");
    printArray(absolutes, size);
    
    // Reduce examples
    printf("Sum of elements: %d\n", reduceArray(arr, size, sum, 0));
    printf("Product of elements: %d\n", reduceArray(arr, size, product, 1));
    printf("Maximum element: %d\n", reduceArray(arr, size, max, arr[0]));
    
    // Combine map and reduce
    printf("Sum of squares: %d\n", reduceArray(squares, size, sum, 0));
    printf("Sum of absolute values: %d\n", reduceArray(absolutes, size, sum, 0));
    
    // Free allocated memory
    free(squares);
    free(doubles);
    free(absolutes);
    
    return 0;
}
```

    return person.weight / (person.height * person.height);

    int count = 0;

        if (filter(people[i])) {
            count++;
        }
    }
    return count;

            result[j++] = people[i];
        }
    }

## Common Pitfalls and Best Practices

### Recursion Pitfalls

1. **Stack Overflow**: Recursive functions can cause stack overflow if there are too many recursive calls or if the base case is not properly defined.

   ```c
   // Dangerous recursion - may cause stack overflow
   int dangerous_recursion(int n) {
       // Missing or incorrect base case
       return n + dangerous_recursion(n - 1);
   }
   ```

2. **Inefficient Repeated Calculations**: As seen in the naive Fibonacci implementation, recursion can lead to many redundant calculations.

3. **Memory Usage**: Each recursive call adds a new layer to the call stack, consuming memory.

### Best Practices for Recursion

1. **Always Define a Proper Base Case**: Ensure there's always a way to terminate the recursion.

   ```c
   int safe_recursion(int n) {
       // Proper base case
       if (n <= 0) {
           return 0;
       }
       return n + safe_recursion(n - 1);
   }
   ```

2. **Consider Tail Recursion**: Tail recursion can be optimized by many compilers.

   ```c
   // Tail-recursive factorial function
   unsigned long factorial_tail(unsigned int n, unsigned long accumulator) {
       if (n == 0 || n == 1) {
           return accumulator;
       }
       return factorial_tail(n - 1, n * accumulator);
   }
   
   // Wrapper function
   unsigned long factorial(unsigned int n) {
       return factorial_tail(n, 1);
   }
   ```

3. **Use Memoization for Expensive Calculations**: Store results of subproblems to avoid redundant calculations.

   ```c
   #include <stdio.h>
   #include <stdlib.h>
   
   // Fibonacci with memoization
   int fibonacci_memo(int n, int* memo) {
       // Base cases
       if (n <= 0) return 0;
       if (n == 1) return 1;
       
       // Check if already computed
       if (memo[n] != -1) {
           return memo[n];
       }
       
       // Compute and store
       memo[n] = fibonacci_memo(n - 1, memo) + fibonacci_memo(n - 2, memo);
       return memo[n];
   }
   
   int fibonacci(int n) {
       // Initialize memoization array
       int* memo = (int*)malloc((n + 1) * sizeof(int));
       for (int i = 0; i <= n; i++) {
           memo[i] = -1;
       }
       
       int result = fibonacci_memo(n, memo);
       free(memo);
       return result;
   }
   ```

4. **Consider Iterative Alternatives**: For some problems, an iterative solution might be more efficient.

### Callback Best Practices

1. **Type Safety**: Use typedefs for callback function pointers to improve clarity and safety.

   ```c
   typedef int (*ComparisonFunction)(int, int);
   
   void bubbleSort(int arr[], int size, ComparisonFunction compare) {
       // Implementation
   }
   ```

2. **Error Handling**: Check if callback functions are valid before using them.

   ```c
   void safeFunction(int arr[], int size, int (*callback)(int)) {
       if (callback == NULL) {
           printf("Error: Null callback function\n");
           return;
       }
       // Use callback
   }
   ```

3. **Context Data**: Pass additional context data to callbacks when needed.

   ```c
   typedef struct {
       // Context data
       int threshold;
       int multiplier;
   } Context;
   
   int transformWithContext(int value, Context* ctx) {
       if (value > ctx->threshold) {
           return value * ctx->multiplier;
       }
       return value;
   }
   
   void processWithContext(int arr[], int size, int (*transform)(int, Context*), Context* ctx) {
       for (int i = 0; i < size; i++) {
           arr[i] = transform(arr[i], ctx);
       }
   }
   ```
